    package environment {
        class Environment {
            +width : int
            +height : int
            +tickTime : int

            -void trySpawn(agent : Agent)
            -boolean inBounds(x : int, y : int)

            +Cell[][] getVisionField(agent : Agent, radius : int)
            +void removeDead()
            +void spawnNewAgents()
            +void tick()
        }
    
        record Config {
            tickTime : int
            width : int
            height : int
            plantsCount : int
            herbivoresCount : int
            predatorsCount : int
        }
        
        record Cell {
            +{static} int getManhattanDistance(x1 : int, y1 : int, x2 : int, y2 : int)
        }
        
        enum CellContent {
            EMPTY
            PLANT
            HERBIVORE
            PREDATOR
            OUT_OF_BOUNDS
            
            {static} CellContent contentOf()
        }
    }
    
    package agents {
        abstract class Agent {
            -alive : boolean;
            #x : int;
            #y : int;
            #int energy = 15;
            +{static} energyLimit : int;
            +{static} offspringCost : int;
        
            +int getX()
            +int getY()
            +int getEnergy()
            +void changeEnergy(delta : int)
            +boolean isAlive()
            +void die()
            +{abstract} Agent makeOffspring(x : int, y : int);
            +{abstract} void act(env : Environment);
        }
        
        class Plant {
            +Agent makeOffspring(x : int, y : int)
            +void act(env : Environment)
        }
        
        abstract class MobileAgent {
            #{abstract} boolean tryEat(env : Environment, victim Agent)
            +{abstract} Direction decideMove(visionField : Cell[][])
            #Direction defineDirection (xStart : int, yStart : int, xDest : int, yDest : int)
            +void setPosition(x: int, y : int)
            +boolean eat(env Environment)
            +void act(env : Environment)
        }
        
        class Herbivore {
            +boolean tryEat(env : Environment, victim : Agent)
            +Agent makeOffspring(x : int, y : int)
            +Direction decideMove(visionField : Cell[][])
        }
        
        class Predator {
            +boolean tryEat(env : Environment, victim : Agent)
            +Agent makeOffspring(x : int, y : int)
            +Direction decideMove(visionField : Cell[][])
        }
        
        enum Direction {
            UP
            DOWN
            RIGHT
            LEFT
        }
    }
